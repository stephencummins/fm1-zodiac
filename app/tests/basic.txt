# Zodiac in the host simulator: ./build.sh --host, then
#   app/build/host/zodiac_host app/tests/basic.txt app/build/host
wait 1500
expect sign 0
expect playing 0
shot aries.png
# the piece sounds by itself
wav aries.wav
wait 4000
wavstop
expect peak_db_min -40
# PRESET steps to the next sign; the music follows
turn PRESET 1
wait 1500
expect sign 1
expect playing 1
expect dirty 1
shot taurus.png
# the knobs: K1 the sign's own, K2 echo
turn K1 3
turn K2 5
wait 100
expect heat 120
expect echo 50
# SAVE keeps the sign and the knobs
tap SAVE
wait 50
expect dirty 0
expect store_writes 1
reboot
wait 1500
expect sign 1
expect heat 120
expect echo 50
# back past Aries wraps to Pisces
turn PRESET -2
wait 1500
expect sign 11
shot pisces.png
# a MIDI program change chooses a sign
midi C4 04 00
wait 1500
expect sign 4
expect playing 4
shot leo.png
wav leo.wav
wait 4000
wavstop
expect peak_db_min -40
# a screen reader hears where it is
listen
wait 100
said ZODIAC. LEO
turn SELECT 1
wait 300
said VIRGO
# PLAY stops the music, the picture goes on; PLAY again and it sounds
tap PLAY
wait 300
expect stopped 1
said STOP
peakreset
wav stopped.wav
wait 1500
wavstop
expect peak_db_max -80
tap PLAY
wait 100
expect stopped 0
said PLAY
peakreset
wav again.wav
wait 4000
wavstop
expect peak_db_min -40
